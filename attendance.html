async function saveAttendance() { clearMessage(); const date = dateInput.value; const selected = getSelectedClass(); if (!date) { showMessage("مهرباني وکړئ د حاضري نېټه انتخاب کړئ.", "error"); return; } if (!selected) { showMessage("مهرباني وکړئ صنف انتخاب کړئ.", "error"); return; } const rows = [ ...document.querySelectorAll("#students .student") ]; if (!rows.length) { showMessage("د ثبت لپاره زده‌کوونکي نشته.", "error"); return; } saveButton.disabled = true; saveButton.textContent = "حاضري ثبتېږي..."; try { // د ټاکلې نېټې او صنف پخوانۍ حاضري ولولئ const existing = await loadExistingAttendance( date, selected.name ); // د زده‌کوونکي ID له مخې پخوانۍ حاضري پیدا کړئ const existingByStudent = new Map(); existing.forEach(record => { const sid = record.student_id ?? record["Student ID"]; if (sid !== null && sid !== undefined) { existingByStudent.set(String(sid), record); } }); const toInsert = []; const updates = []; rows.forEach(row => { const studentId = Number(row.dataset.studentId); const status = row.dataset.status || "حاضر"; // دواړه بڼې ډکوو، ځکه په جدول کې دواړه کالمونه شته const record = { "Date": date, "Student ID": studentId, "Class": selected.name, "Status": status, date: date, student_id: studentId, class: selected.name, status: status }; const oldRecord = existingByStudent.get( String(studentId) ); if (oldRecord && oldRecord.id != null) { updates.push({ id: oldRecord.id, data: record }); } else { toInsert.push(record); } }); // پخوانۍ حاضري تازه کړئ؛ حذف یې مه کوئ for (const item of updates) { const response = await fetch( `${ATTENDANCE_API}?id=eq.${encodeURIComponent(item.id)}`, { method: "PATCH", headers: { ...headers, "Prefer": "return=representation" }, body: JSON.stringify(item.data) } ); if (!response.ok) { throw new Error( "د پخوانۍ حاضري تازه کول ناکام شول:\n" + await response.text() ); } } // نوې حاضري ثبت کړئ if (toInsert.length > 0) { const response = await fetch( ATTENDANCE_API, { method: "POST", headers: { ...headers, "Prefer": "return=representation" }, body: JSON.stringify(toInsert) } ); if (!response.ok) { throw new Error( "د نوې حاضري ثبت ناکام شو:\n" + await response.text() ); } } showMessage( `✅ د حاضري ثبت بریالی شو!\n` + `صنف: ${selected.name}\n` + `نېټه: ${date}\n` + `ټول زده‌کوونکي: ${rows.length}\n` + `نوي ریکارډونه: ${toInsert.length}\n` + `تازه شوي ریکارډونه: ${updates.length}`, "success" ); countDiv.textContent = `ټول زده‌کوونکي: ${rows.length} | حاضري ثبت شوې`; } catch (error) { console.error(error); showMessage( "❌ د حاضري ثبتولو ستونزه:\n" + error.message, "error" ); } finally { saveButton.disabled = false; saveButton.textContent = "💾 د ټول صنف حاضري ثبت کړئ"; } }
